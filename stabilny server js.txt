const express = require("express");
const http = require("http");
const path = require("path");
const { Server } = require("socket.io");

const app = express();
const server = http.createServer(app);

/* =========================
   SOCKET.IO
========================= */

const io = new Server(server, {
  cors: {
    origin: "*",
    methods: ["GET", "POST"]
  }
});


/* =========================
   PLIKI PUBLIC
========================= */

app.use(express.static(path.join(__dirname, "public")));


/* =========================
   KOLEJKA
========================= */

let waitingUsers = [];


/* =========================
   SPRAWDZANIE DOPASOWANIA
========================= */

function canMatch(a, b) {

  if (!a || !b) {
    return false;
  }


  const aOK =
    a.seeking === "random" ||
    a.seeking === b.gender;


  const bOK =
    b.seeking === "random" ||
    b.seeking === a.gender;


  if (!aOK || !bOK) {
    return false;
  }


  const aAgeOK =
    b.age >= a.minAge &&
    b.age <= a.maxAge;


  const bAgeOK =
    a.age >= b.minAge &&
    a.age <= b.maxAge;


  return aAgeOK && bAgeOK;
}


/* =========================
   LICZBA ONLINE
========================= */

function getOnlineCount() {

  return io.sockets.sockets.size;
}


function updateOnline() {

  const count =
    getOnlineCount();


  io.emit(
    "online",
    count
  );


  console.log(
    "Online:",
    count
  );
}


/* =========================
   WALIDACJA GIF
========================= */

function getValidProfileGif(value) {

  if (
    typeof value !== "string" ||
    value.length === 0
  ) {
    return null;
  }


  if (value.length > 1000) {
    return null;
  }


  try {

    const url =
      new URL(value);


    if (
      url.protocol !== "https:"
    ) {
      return null;
    }


    if (
      !url.hostname.includes(
        "giphy.com"
      )
    ) {
      return null;
    }


    return value;

  } catch {

    return null;
  }
}


/* =========================
   USUWANIE Z KOLEJKI
========================= */

function removeFromQueue(socketId) {

  waitingUsers =
    waitingUsers.filter(
      user =>
        user &&
        user.socket &&
        user.socket.id !== socketId
    );
}


/* =========================
   CZYSZCZENIE KOLEJKI
========================= */

function cleanWaitingUsers() {

  waitingUsers =
    waitingUsers.filter(
      user => {

        if (
          !user ||
          !user.socket
        ) {
          return false;
        }


        const connected =
          io.sockets.sockets.has(
            user.socket.id
          );


        if (!connected) {
          return false;
        }


        if (
          user.socket.data.state !==
          "waiting"
        ) {
          return false;
        }


        return true;
      }
    );
}


/* =========================
   SOCKET.IO
========================= */

io.on(
  "connection",
  (socket) => {

    console.log(
      "=============================="
    );

    console.log(
      "POŁĄCZONO:",
      socket.id
    );

    console.log(
      "IP:",
      socket.handshake.address
    );

    console.log(
      "Online:",
      getOnlineCount()
    );

    console.log(
      "=============================="
    );


    /* =========================
       DOMYŚLNY PROFIL
    ========================= */

    socket.data.gender =
      null;


    socket.data.seeking =
      "random";


    socket.data.age =
      25;


    socket.data.minAge =
      18;


    socket.data.maxAge =
      80;


    socket.data.profileGif =
      null;


    socket.data.room =
      null;


    socket.data.state =
      "idle";


    /* =========================
       ONLINE DLA NOWEGO
    ========================= */

    socket.emit(
      "online",
      getOnlineCount()
    );


    updateOnline();


    /* =========================
       USTAWIANIE PROFILU
    ========================= */

    socket.on(
      "setProfile",
      (data) => {

        if (!data) {
          return;
        }


        socket.data.gender =
          data.gender || null;


        socket.data.seeking =
          data.seeking || "random";


        socket.data.age =
          Number(data.age) || 25;


        socket.data.minAge =
          Number(data.minAge) || 18;


        socket.data.maxAge =
          Number(data.maxAge) || 80;


        /* =========================
           WIEK
        ========================= */

        if (
          socket.data.age < 18
        ) {
          socket.data.age = 18;
        }


        if (
          socket.data.age > 80
        ) {
          socket.data.age = 80;
        }


        /* =========================
           MIN WIEK
        ========================= */

        if (
          socket.data.minAge < 18
        ) {
          socket.data.minAge = 18;
        }


        if (
          socket.data.minAge > 80
        ) {
          socket.data.minAge = 80;
        }


        /* =========================
           MAX WIEK
        ========================= */

        if (
          socket.data.maxAge < 18
        ) {
          socket.data.maxAge = 18;
        }


        if (
          socket.data.maxAge > 80
        ) {
          socket.data.maxAge = 80;
        }


        /* =========================
           MIN > MAX
        ========================= */

        if (
          socket.data.minAge >
          socket.data.maxAge
        ) {

          const temp =
            socket.data.minAge;


          socket.data.minAge =
            socket.data.maxAge;


          socket.data.maxAge =
            temp;
        }


        /* =========================
           GIF
        ========================= */

        socket.data.profileGif =
          getValidProfileGif(
            data.profileGif
          );


        console.log(
          "Profil:",
          socket.id,
          "| płeć:",
          socket.data.gender,
          "| szuka:",
          socket.data.seeking,
          "| wiek:",
          socket.data.age,
          "| przedział:",
          socket.data.minAge +
            "-" +
            socket.data.maxAge,
          "| GIF:",
          socket.data.profileGif
            ? "TAK"
            : "NIE"
        );


        socket.emit(
          "online",
          getOnlineCount()
        );
      }
    );


    /* =========================
       SZUKANIE PARTNERA
    ========================= */

    socket.on(
      "find",
      () => {

        console.log(
          "Szukanie:",
          socket.id,
          "| stan:",
          socket.data.state
        );


        /* =========================
           BLOKADA PODWÓJNEGO FIND
        ========================= */

        if (
          socket.data.state ===
          "waiting"
        ) {

          console.log(
            "FIND IGNORE - już czeka:",
            socket.id
          );

          return;
        }


        if (
          socket.data.state ===
          "matched"
        ) {

          console.log(
            "FIND IGNORE - już ma rozmowę:",
            socket.id
          );

          return;
        }


        /* =========================
           USUŃ STARY WPIS
        ========================= */

        removeFromQueue(
          socket.id
        );


        cleanWaitingUsers();


        /* =========================
           SZUKANIE
        ========================= */

        const matchIndex =
          waitingUsers.findIndex(
            user => {

              if (
                !user ||
                !user.socket
              ) {
                return false;
              }


              if (
                user.socket.id ===
                socket.id
              ) {
                return false;
              }


              if (
                user.socket.data.state !==
                "waiting"
              ) {
                return false;
              }


              return canMatch(
                socket.data,
                user.socket.data
              );
            }
          );


        /* =========================
           ZNALEZIONO PARTNERA
        ========================= */

        if (
          matchIndex !== -1
        ) {

          const partnerObj =
            waitingUsers.splice(
              matchIndex,
              1
            )[0];


          const partner =
            partnerObj.socket;


          /* =========================
             SPRAWDZENIE PARTNERA
          ========================= */

          if (
            !partner ||
            !io.sockets.sockets.has(
              partner.id
            )
          ) {

            socket.data.state =
              "waiting";


            waitingUsers.push({
              socket: socket
            });


            socket.emit(
              "waiting"
            );


            return;
          }


          if (
            partner.data.state !==
            "waiting"
          ) {

            socket.data.state =
              "waiting";


            waitingUsers.push({
              socket: socket
            });


            socket.emit(
              "waiting"
            );


            return;
          }


          /* =========================
             USUŃ Z KOLEJKI
          ========================= */

          removeFromQueue(
            socket.id
          );


          removeFromQueue(
            partner.id
          );


          /* =========================
             POKÓJ
          ========================= */

          const room =
            "room_" +
            Date.now() +
            "_" +
            socket.id;


          socket.join(room);
          partner.join(room);


          socket.data.room =
            room;


          partner.data.room =
            room;


          socket.data.state =
            "matched";


          partner.data.state =
            "matched";


          /* =========================
             MATCH
          ========================= */

          socket.emit(
            "matched"
          );


          partner.emit(
            "matched"
          );


          /* =========================
             PROFIL PARTNERA
          ========================= */

          socket.emit(
            "partner-profile",
            {
              profileGif:
                partner.data.profileGif ||
                null
            }
          );


          partner.emit(
            "partner-profile",
            {
              profileGif:
                socket.data.profileGif ||
                null
            }
          );


          /* =========================
             CONNECTED
          ========================= */

          socket.emit(
            "connected"
          );


          partner.emit(
            "connected"
          );


          console.log(
            "=============================="
          );

          console.log(
            "MATCH:",
            socket.id,
            "<->",
            partner.id
          );

          console.log(
            "Pokój:",
            room
          );

          console.log(
            "=============================="
          );

        } else {

          /* =========================
             BRAK PARTNERA
          ========================= */

          socket.data.state =
            "waiting";


          waitingUsers.push({
            socket: socket
          });


          socket.emit(
            "waiting"
          );


          console.log(
            "Czeka:",
            socket.id,
            "| kolejka:",
            waitingUsers.length
          );
        }


        updateOnline();
      }
    );


    /* =========================
       WIADOMOŚCI
    ========================= */

    socket.on(
      "message",
      (text) => {

        const room =
          socket.data.room;


        if (!room) {
          return;
        }


        if (
          typeof text !== "string"
        ) {
          return;
        }


        if (
          text.length > 5000
        ) {
          return;
        }


        io.to(room).emit(
          "message",
          {
            text,
            sender:
              socket.id
          }
        );
      }
    );


    /* =========================
       PISANIE
    ========================= */

    socket.on(
      "typing",
      () => {

        const room =
          socket.data.room;


        if (!room) {
          return;
        }


        socket
          .to(room)
          .emit(
            "typing"
          );
      }
    );


    /* =========================
       RĘCZNE ROZŁĄCZENIE
    ========================= */

    socket.on(
      "manual-disconnect",
      () => {

        console.log(
          "Manual disconnect:",
          socket.id
        );


        removeFromQueue(
          socket.id
        );


        const room =
          socket.data.room;


        /* =========================
           BRAK ROZMOWY
        ========================= */

        if (!room) {

          socket.data.state =
            "idle";


          socket.emit(
            "disconnected"
          );


          return;
        }


        /* =========================
           ZNAJDŹ PARTNERA
        ========================= */

        const partner =
          Array.from(
            io.sockets.sockets.values()
          ).find(
            other =>
              other.id !== socket.id &&
              other.data.room === room
          );


        /* =========================
           PARTNER
        ========================= */

        if (partner) {

          partner.data.room =
            null;


          partner.data.state =
            "idle";


          partner.emit(
            "partner-left"
          );


          console.log(
            "Partner został poinformowany:",
            partner.id
          );
        }


        /* =========================
           OPUŚĆ POKÓJ
        ========================= */

        socket.leave(
          room
        );


        socket.data.room =
          null;


        socket.data.state =
          "idle";


        socket.emit(
          "disconnected"
        );


        console.log(
          "Manual disconnect zakończony:",
          socket.id
        );
      }
    );


    /* =========================
       DISCONNECT
    ========================= */

    socket.on(
      "disconnect",
      (reason) => {

        console.log(
          "=============================="
        );

        console.log(
          "ROZŁĄCZONO:",
          socket.id
        );

        console.log(
          "Powód:",
          reason
        );

        console.log(
          "=============================="
        );


        removeFromQueue(
          socket.id
        );


        const room =
          socket.data.room;


        /* =========================
           PARTNER
        ========================= */

        if (room) {

          const partner =
            Array.from(
              io.sockets.sockets.values()
            ).find(
              other =>
                other.id !== socket.id &&
                other.data.room === room
            );


          if (partner) {

            partner.data.room =
              null;


            partner.data.state =
              "idle";


            partner.emit(
              "partner-left"
            );


            console.log(
              "Partner po rozłączeniu:",
              partner.id
            );
          }
        }


        updateOnline();


        console.log(
          "Kolejka:",
          waitingUsers.length
        );
      }
    );
  }
);
