seconds=
Math.max(
0,
Math.floor(
Number(seconds)||0
)
);

const minutes=
Math.floor(
seconds/60
);

const remaining=
seconds%60;

String(minutes).padStart(2,"0")+
":"+
String(remaining).padStart(2,"0")
);

}
